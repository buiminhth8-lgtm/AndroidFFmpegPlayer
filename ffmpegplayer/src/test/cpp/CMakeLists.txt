cmake_minimum_required(VERSION 3.22)
project(recorder_host_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
enable_testing()
find_package(Threads REQUIRED)
set(PLAYER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../main/cpp" CACHE PATH "Player native source directory")
add_executable(RecorderAsyncTest RecorderAsyncTest.cpp recorder_fakes/FakeFFmpeg.cpp
    "${PLAYER_SOURCE_DIR}/native/PlayerRemuxRecorder.cpp")
target_include_directories(RecorderAsyncTest PRIVATE recorder_fakes "${PLAYER_SOURCE_DIR}" "${PLAYER_SOURCE_DIR}/ffmpeg/include")
target_link_libraries(RecorderAsyncTest PRIVATE Threads::Threads)
option(RECORDER_ASAN "Enable AddressSanitizer for recorder ownership tests" OFF)
if(RECORDER_ASAN)
    if(MSVC)
        target_compile_options(RecorderAsyncTest PRIVATE /fsanitize=address /Zi)
        target_link_options(RecorderAsyncTest PRIVATE /INCREMENTAL:NO)
    else()
        target_compile_options(RecorderAsyncTest PRIVATE -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(RecorderAsyncTest PRIVATE -fsanitize=address)
    endif()
endif()
if(MSVC)
    target_compile_options(RecorderAsyncTest PRIVATE /utf-8 /W4)
    target_compile_definitions(RecorderAsyncTest PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME RecorderAsyncTest COMMAND RecorderAsyncTest)
set_tests_properties(RecorderAsyncTest PROPERTIES TIMEOUT 45)
foreach(test DiagnosticsModeTest E2ETimebaseTest PreT0TimingTrackerTest)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${test}.cpp")
        add_executable(${test} "${test}.cpp")
        target_link_libraries(${test} PRIVATE Threads::Threads)
        add_test(NAME ${test} COMMAND ${test})
    endif()
endforeach()
foreach(enabled 0 1)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/TestHookPolicyTest.cpp")
        add_executable(TestHookPolicyTest${enabled} TestHookPolicyTest.cpp)
        target_compile_definitions(TestHookPolicyTest${enabled} PRIVATE FFMPEGPLAYER_ENABLE_TEST_HOOKS=${enabled} FFMPEGPLAYER_EXPECT_TEST_HOOKS=${enabled})
        add_test(NAME TestHookPolicyTest${enabled} COMMAND TestHookPolicyTest${enabled})
    endif()
endforeach()
